  title       = {Lagrangian Putt-Putt: A Holistic Investigation into Holonomically Constrained Forced-Geodesic Motion on Arbitrary $C^2$-Surfaces},

  year        = {PHDS'26},
  affil       = {UgS III},
  category    = {article},
  abstract    = {This paper concerns the holistic investigation of golf ball dynamics.
		Specifically, in this paper we seek to determine the generalized equations
		of motion for a golf ball constained to a golf course of the form
		$z=h(x,y)$ under constant gravity $g$. In doing so, concepts such as
		Lagrangian mechanics, the principle of least action, variational calculus,
		and differential geometry are brought up. The paper is written in such a
		way that a reader with basic calculus knowledge should be able to
		meaningfully extract the important information of this paper.
  },
